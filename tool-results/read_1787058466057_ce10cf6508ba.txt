  1200→function animateCounter(el, from, to, duration) {
  1201→  const start = performance.now();
  1202→  function easeInOutCubic(t) {
  1203→    return t < 0.5 ? 4 * t * t * t : 1 - Math.pow(-2 * t + 2, 3) / 2;
  1204→  }
  1205→  function tick(now) {
  1206→    const elapsed = now - start;
  1207→    const progress = Math.min(elapsed / duration, 1);
  1208→    const value = Math.round(from + (to - from) * easeInOutCubic(progress));
  1209→    el.textContent = String(value).padStart(3, '0');
  1210→    if (progress < 1) requestAnimationFrame(tick);
  1211→  }
  1212→  requestAnimationFrame(tick);
  1213→}

  1373→/* ── Word Reveal ── */
  1374→function setupWordReveal() {
  1375→  const container = document.getElementById('aboutText');
  1376→  if (!container) return;
  1377→  // Get the HTML, split text nodes into words wrapped in spans
  1378→  const walker = document.createTreeWalker(container, NodeFilter.SHOW_TEXT, null, false);
  1379→  const textNodes = [];
  1380→  while (walker.nextNode()) textNodes.push(walker.currentNode);
  1381→
  1382→  let wordIndex = 0;
  1383→  textNodes.forEach(node => {
  1384→    const text = node.textContent;
  1385→    const words = text.split(/(\s+)/);
  1386→    const frag = document.createDocumentFragment();
  1387→    words.forEach(part => {
  1388→      if (/^\s+$/.test(part)) {
  1389→        frag.appendChild(document.createTextNode(part));
  1390→      } else if (part.length > 0) {
  1391→        const span = document.createElement('span');
  1392→        span.className = 'word-reveal-word';
  1393→        span.textContent = part;
  1394→        span.style.transitionDelay = (wordIndex * 35) + 'ms';
  1395→        frag.appendChild(span);
  1396→        wordIndex++;
  1397→      }
  1398→    });
  1399→    node.parentNode.replaceChild(frag, node);
  1400→  });
  1401→
  1402→  // Observe for visibility
  1403→  const observer = new IntersectionObserver((entries) => {
  1404→    entries.forEach(entry => {
  1405→      if (entry.isIntersecting) {
  1406→        container.querySelectorAll('.word-reveal-word').forEach(w => w.classList.add('visible'));
  1407→        observer.unobserve(container);
  1408→      }
  1409→    });
  1410→  }, { threshold: 0.2 });
  1411→  observer.observe(container);
  1412→}
  1413→setupWordReveal();
